  # PostgreSQL Database
  postgres:
    image: pgvector/pgvector:pg16
    container_name: aquatiq-postgres
    restart: unless-stopped
    environment:
      POSTGRES_DB: aquatiq_hub
      POSTGRES_USER: aquatiq_user
      POSTGRES_PASSWORD: aquatiq_password
    ports:
      - "5432:5432"
    volumes:
      - postgres_data:/var/lib/postgresql/data
      - ./docker/postgres/init.sql:/docker-entrypoint-initdb.d/init.sql
    networks:
      - aquatiq-network
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U aquatiq_user -d aquatiq_hub"]
      interval: 10s
      timeout: 5s
      retries: 5

    healthcheck:

      timeout: 5s
      retries: 5
    command: redis-server --appendonly yes
