'use client';

import React, { useState, useCallback } from 'react';
import { 
  Mail
} from 'lucide-react';

// Import our modular system components
import { useAuth } from './hooks/use-auth';
import { useAuthForm } from './hooks/use-auth-form';
import { usePasskey } from './hooks/use-passkey';
// Import Better Auth client
import { authClient } from './lib/auth-client-enterprise';
// Consent: using lightweight hook (keys: necessary|performance|functional|marketing)
// Enterprise consent hook (advanced categories incl. analytics & social)
import { useConsent } from './consent/useConsent';
import { useTwoFactor } from './hooks/useTwoFactor';
import { useVerification } from './hooks/useVerification';
import { useSecuritySettings } from './hooks/useSecuritySettings';

// Import i18n hooks
// i18n system (standardized path alias)
import { useTranslation, useLanguageSwitch } from './lib/i18n/hooks';

// New hooks
// import { useLoading } from '../../hooks/use-loading';
// import { useToast, useToastActions } from '../../hooks/use-toast';
// import { useI18n } from '../../hooks/use-i18n';

// UI Components
import { AuthTabs } from './core/AuthTabs';
import { AuthForms } from './core/AuthForms';
import { SocialProviders } from './providers/SocialProviders';
import { PasskeyButtons } from './providers/PasskeyButtons';
import { OrganizationManagement } from './providers/OrganizationManagement';
import { AuthIllustration } from './ui/AuthIllustration';
import { LanguageSwitcher } from './ui/LanguageSwitcher';
import { Logo } from './ui';
import { AuthGuard } from './guards/AuthGuard';
import { RoleGuard } from './guards/RoleGuard';
import { ConsentPreferences } from './consent/ConsentPreferences';
import { CallbackModal } from './modals/CallbackModal';

// New UI Components
// import { ToastViewport } from '../ui/toast';
// import { LoadingOverlay } from '../ui/loading-components';

// Types
import { AuthMode } from './types/auth';

// Default values when enhanced features are not available
const defaultToastActions = {
  error: (title: string, description?: string) => console.log('Error:', title, description),
  info: (message: string) => console.log('Info:', message),
};

export interface AuthPageProps {
  redirectTo?: string;
  initialMode?: AuthMode;
  features?: {
    socialAuth?: boolean;
    passkeys?: boolean;
    enterpriseSSO?: boolean;
    organizationManagement?: boolean;
    twoFactor?: boolean;
    multiChannelVerification?: boolean;
    gdprConsent?: boolean;
    accessibilityMode?: boolean;
  };
  customization?: {
    theme?: 'light' | 'dark' | 'auto';
    primaryColor?: string;
    logo?: string;
    companyName?: string;
    supportEmail?: string;
  };
  onAuthSuccess?: (user: unknown) => void;
  onAuthError?: (error: Error) => void;
  onModeChange?: (mode: AuthMode) => void;
}

export function AuthPage({
  initialMode = 'signin',
  features = {
    socialAuth: true,
    passkeys: true,
    enterpriseSSO: true,
    organizationManagement: true,
    twoFactor: true,
    multiChannelVerification: true,
    gdprConsent: true,
    accessibilityMode: true
  },
  customization = {
    theme: 'auto',
    companyName: 'ID-Knuten',
    supportEmail: 'support@id-knuten.no'
  },
  onAuthSuccess,
  onAuthError,
  onModeChange
}: AuthPageProps) {
  
  // Core hooks
  const { isLoading: authLoading } = useAuth();
  const { 
    formData, 
    errors, 
    isLoading: formLoading,
    setField,
    handleSubmit,
    reset
  } = useAuthForm();
  
  // i18n hooks
  const { t } = useTranslation();
  const { isNorwegian } = useLanguageSwitch();
  
  // Specialized hooks  
  const { 
    isSupported: passkeySupported,
    isLoading: passkeyLoading
  } = usePasskey();
  
  const {
    consent,
    showBanner: showConsentBanner,
    showPreferences: consentPreferencesOpen,
    acceptAll: acceptAllConsent,
    rejectAll: rejectAllConsent,
    showPreferencesModal: openConsentPreferences,
    hidePreferences: hideConsentPreferences,
    updateConsent,
    saveConsent,
  } = useConsent();
  
  const {
    isLoading: twoFactorLoading
  } = useTwoFactor();
  
  const {
    isLoading: verificationLoading
  } = useVerification();
  
  const {
    isLoading: securityLoading
  } = useSecuritySettings();

  // New enhanced hooks - using defaults for now (can be upgraded to full providers later)
  const { error, info } = defaultToastActions;

  // Local state
  const [currentMode, setCurrentMode] = useState<AuthMode>(initialMode);
  const [showCallbackModal, setShowCallbackModal] = useState(false);

  // Check for callback scenario in URL
  React.useEffect(() => {
    if (typeof window !== 'undefined') {
      const urlParams = new URLSearchParams(window.location.search);
      const isCallback = urlParams.has('code') || urlParams.has('state') || window.location.pathname.includes('/callback');
      
      if (isCallback) {
        setShowCallbackModal(true);
      }
    }
  }, []);

  // Handle mode changes
  const handleModeChange = useCallback((newMode: AuthMode) => {
    setCurrentMode(newMode);
    onModeChange?.(newMode);
    reset();
  }, [onModeChange, reset]);

  // Authentication handlers
  const handleSocialSignIn = async (provider: string) => {
    try {
      // Validate provider
      if (provider !== 'microsoft' && provider !== 'google') {
        throw new Error(`Unsupported provider: ${provider}`);
      }

      const redirectTo = `${window.location.origin}/auth/callback`;
      const endpoint = `${window.location.origin}/api/auth/oauth/initiate`;

      console.log(`🔄 Initiating ${provider} sign-in via Better Auth...`);

      const response = await fetch(endpoint, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        credentials: 'include',
        redirect: 'manual',
        body: JSON.stringify({
          provider,
          redirectTo,
        }),
      });

      if (!response.ok) {
        const errorText = await response.text();
        console.error('OAuth initiation failed:', errorText);
        throw new Error(`Failed to initiate social sign-in: ${response.statusText}`);
      }

      const data = await response.json().catch(() => ({}));
      if (data?.url) {
        window.location.href = data.url;
        return;
      }

      throw new Error('No redirect URL received from authentication service');
      
    } catch (err) {
      console.error('Social sign-in error:', err);
      onAuthError?.(err instanceof Error ? err : new Error('Social sign-in failed'));
    }
  };

  // Loading state - combine all loading states
  const isLoading = authLoading || formLoading || passkeyLoading || twoFactorLoading || verificationLoading || securityLoading;

  // Get header text based on mode using i18n
  const getHeaderText = () => {
    switch (currentMode) {
      case 'signup':
        return {
          title: t('auth.modes.signup.title'),
          description: t('auth.modes.signup.description')
        };
      case 'signin':
        return {
          title: t('auth.modes.signin.title'),
          description: t('auth.modes.signin.description')
        };
      case 'enterprise-sso':
        return {
          title: t('auth.sso.title'),
          description: t('auth.sso.description')
        };
      case 'org':
        return {
          title: t('auth.organizationManagement.title'),
          description: t('auth.organizationManagement.description')
        };
      default:
        return {
          title: t('auth.modes.default.title'),
          description: t('auth.modes.default.description')
        };
    }
  };

  const headerText = getHeaderText();

  // Localized UI strings not yet in central translation catalog (co-located for now)
  const uiTexts = {
    termsPrefixSignin: t('auth.page.terms.prefixSignin' as string),
    termsAnd: t('auth.page.terms.and' as string),
    termsUser: t('auth.page.terms.termsOfUse' as string),
    termsPrivacy: t('auth.page.terms.privacyPolicy' as string),
    signupDataNotice: t('auth.page.terms.dataNotice' as string),
    deleteCookie: t('auth.page.terms.deleteCookie' as string),
    supportNeedHelp: t('auth.page.support.needHelp' as string),
    supportContact: t('auth.page.support.contactSupport' as string),
    helpLabel: t('auth.page.support.helpLabel' as string),
    imprint: isNorwegian ? 'Om oss' : 'Imprint',
    privacy: isNorwegian ? 'Personvern' : 'Privacy',
    copyright: isNorwegian ? 'Opphavsrett' : 'Copyright',
    cookieSettings: isNorwegian ? 'Cookie-innstillinger' : 'Cookie settings'
  } as const;

  return (
  <div className="h-screen w-screen flex flex-col items-center justify-center p-2 sm:p-3 md:p-4 lg:p-6 overflow-hidden">
      {/* Main Auth Container - More compact design */}
      <div className="bg-card rounded-lg sm:rounded-xl lg:rounded-2xl shadow-lg lg:shadow-xl overflow-hidden w-full max-w-xs sm:max-w-sm md:max-w-3xl lg:max-w-4xl xl:max-w-5xl flex flex-col">

        <div className="flex flex-col lg:flex-row min-h-full w-full overflow-hidden">
          
          {/* Left Panel - Authentication Forms - Always visible */}
          <div className="w-full lg:w-2/5 xl:w-1/3 p-3 sm:p-4 md:p-5 lg:p-6 flex flex-col justify-center lg:relative lg:left-6 xl:left-8 items-center bg-background min-h-full overflow-y-auto overflow-x-hidden">
            <div className="max-w-xs sm:max-w-sm mx-auto w-full flex flex-col h-full justify-center py-3 min-h-0">
              

            {/* Logo - Above Auth Tabs */}
              <div className="relative left-5 flex justify-left mb-4">
                <Logo />
              </div>

              {/* Norwegian Tabs Navigation - Responsive */}
              <div className="flex flex-col w-full items-center justify-center">
                <AuthTabs 
                  currentMode={currentMode}
                  onModeChange={handleModeChange}
                  disabled={isLoading}
                  features={{
                    enterpriseSSO: features.enterpriseSSO,
                    organizationManagement: features.organizationManagement
                  }}
                />
              </div>

              {/* Norwegian Header Text - Responsive sizing */}
              <div className="mb-3 text-center w-full">
                <h1 className="text-base md:text-lg lg:text-xl font-semibold text-foreground mb-1 text-center">
                  {headerText.title}
                </h1>
                <p className="text-sm text-muted-foreground leading-relaxed text-center mx-auto max-w-full">
                  {headerText.description}
                </p>
              </div>

              {/* Main Auth Forms/Organization Management - Compact on mobile */}
              <div className="mb-3">
                {currentMode === 'org' ? (
                  <OrganizationManagement 
                    onSuccess={(result) => {
                      console.log('Organization action success:', result);
                      onAuthSuccess?.(result);
                    }}
                    onError={(error) => {
                      console.error('Organization error:', error);
                      onAuthError?.(new Error(error));
                    }}
                  />
                ) : (
                  <AuthForms 
                    mode={currentMode}
                    formData={formData}
                    errors={errors}
                    isLoading={isLoading}
                    onFieldChange={setField}
                    onSubmit={handleSubmit}
                  />
                )}
              </div>

              {/* Social Authentication - Always visible, responsive sizing */}
              {features.socialAuth && (currentMode === 'signin' || currentMode === 'signup') && (
                <div className="mb-3">
                  <SocialProviders 
                    onProviderClick={handleSocialSignIn}
                    isLoading={isLoading}
                  />
                </div>
              )}

              {/* Passkey Authentication - Essential feature, always visible but compact */}
              {features.passkeys && passkeySupported && currentMode !== 'org' && (
                <div className="mb-3">
                  <PasskeyButtons 
                    mode={currentMode === 'signin' ? 'authenticate' : currentMode === 'signup' ? 'register' : 'manage'}
                    onSuccess={() => {}}
                    onError={(error) => onAuthError?.(new Error(error))}
                    email={formData.email}
                    displayName={formData.name}
                    disabled={isLoading}
                    className="text-sm"
                  />
                </div>
              )}

              {/* Mode-specific links - Compact on mobile */}
              <div className="text-center space-y-1 mb-2">
                {/* Forgot password flow not in current AuthMode union; add when implemented */}
              </div>

              {/* GDPR text - Norwegian - Hidden on small mobile, compact on others */}
              <div className="hidden xs:block text-center mb-2">
                {currentMode === 'signin' && (
                  <p className="text-xs text-muted-foreground leading-relaxed mb-1 px-2" aria-label="terms-and-privacy">
                    {uiTexts.termsPrefixSignin}{' '}
                    <button type="button" className="text-primary hover:text-primary/80 underline focus:outline-none">
                      {uiTexts.termsUser}
                    </button>{' '}{uiTexts.termsAnd}{' '}
                    <button type="button" className="text-primary hover:text-primary/80 underline focus:outline-none">
                      {uiTexts.termsPrivacy}
                    </button>.
                  </p>
                )}
                {currentMode === 'signup' && (
                  <p className="text-xs text-muted-foreground leading-relaxed px-2" aria-label="signup-data-notice">
                    {uiTexts.signupDataNotice}{' '}
                    <button 
                      type="button"
                      className="text-primary underline hover:opacity-90 focus:outline-none"
                      onClick={() => {
                        // Cookie deletion logic
                        document.cookie.split(";").forEach(function(c) { 
                          document.cookie = c.replace(/^ +/, "").replace(/=.*/, "=;expires=" + new Date().toUTCString() + ";path=/"); 
                        });
                        localStorage.clear();
                        sessionStorage.clear();
                      }}
                    >
                      {uiTexts.deleteCookie}
                    </button>.
                  </p>
                )}
              </div>

              {/* Support - Hidden on mobile, visible on tablet+ */}
              <div className="hidden sm:block mt-3 text-center">
                <p className="text-xs text-muted-foreground">
                  {uiTexts.supportNeedHelp}{' '}
                  <a 
                    href={`mailto:${customization.supportEmail}`}
                    className="text-primary hover:text-primary/80 transition-colors"
                  >
                    {uiTexts.supportContact}
                  </a>
                </p>
              </div>

              {/* Mobile-only quick actions */}
              <div className="sm:hidden mt-1 flex justify-center space-x-4">
                {!showConsentBanner && (
                  <a 
                    href={`mailto:${customization.supportEmail}`}
                    className="text-xs text-muted-foreground hover:text-foreground transition-colors flex items-center gap-1"
                    aria-label={uiTexts.helpLabel}
                  >
                    <Mail className="w-3 h-3" />
                    {uiTexts.helpLabel}
                  </a>
                )}
              </div>
            </div>
          </div>

          {/* Right Panel - Illustration - Progressive enhancement */}
          {/* Hidden on mobile, compact on tablet, full on desktop */}
          <div className="hidden md:flex w-full md:w-1/2 lg:w-3/5 xl:w-2/3 relative overflow-hidden items-center justify-center p-2 md:p-3 lg:p-4 min-h-0">
            <AuthIllustration 
              companyName={customization.companyName}
              showConsentBanner={features.gdprConsent && showConsentBanner}
              onAcceptAll={acceptAllConsent}
              onRejectAll={rejectAllConsent}
              onShowPreferences={openConsentPreferences}
              className="w-full h-full max-w-full max-h-full"
            />
          </div>
        </div>

      </div>

      {/* Standalone Footer (separated from modal) */}
      <div className="absolute bottom-5 md:bottom-1 text-center px-5">
        <div className="flex flex-wrap justify-center gap-8 mb-3">
          <button className="text-xs text-muted-foreground hover:text-foreground transition-colors">
            {uiTexts.imprint}
          </button>
          <button className="text-xs text-muted-foreground hover:text-foreground transition-colors">
            {uiTexts.privacy}
          </button>
          <button className="text-xs text-muted-foreground hover:text-foreground transition-colors">
            {uiTexts.copyright}
          </button>
          <button 
            className="text-xs text-muted-foreground hover:text-foreground transition-colors"
            onClick={openConsentPreferences}
          >
            {uiTexts.cookieSettings}
          </button>


        {/* Language Switcher - Above Auth Tabs */}
          <div className="relative left-5 flex justify-left mb-4">
            <LanguageSwitcher 
              variant="dropdown" 
              size="sm"
              className="opacity-80 hover:opacity-100 transition-opacity"
            />
          </div>
        </div>
        <p className="text-xs text-muted-foreground/70">
          Copyright © 2025 {customization.companyName} Norway
        </p>
      </div>

      {/* Consent Preferences Modal */}
      <ConsentPreferences
        isOpen={consentPreferencesOpen}
        onClose={() => hideConsentPreferences()}
        consent={consent}
        onConsentChange={updateConsent}
        onSave={() => saveConsent(consent)}
        onAcceptAll={acceptAllConsent}
        onRejectAll={rejectAllConsent}
      />

      {/* Social Auth Callback Modal */}
      <CallbackModal
        isOpen={showCallbackModal}
        onClose={() => setShowCallbackModal(false)}
      />

      {/* Enhanced features - uncomment when providers are added */}
      {/* Global Loading Overlay */}
      {/*<LoadingOverlay
        isVisible={globalLoading.isLoading}
        message={globalLoading.message}
        stage={globalLoading.stage}
        progress={globalLoading.progress}
      />*/}

      {/* Toast Notifications */}
      {/*<ToastViewport
        toasts={toasts}
        onDismiss={dismissToast}
      />*/}
    </div>
  );
}

// Export component with guards for protected routes
export const ProtectedAuthPage = () => (
  <AuthGuard allowedStates={['unauthenticated']}>
    <AuthPage />
  </AuthGuard>
);

// Export role-based auth page
export const AdminAuthPage = () => (
  <RoleGuard allowedRoles={['admin']}>
    <AuthPage features={{ organizationManagement: true }} />
  </RoleGuard>
);

export default AuthPage;
